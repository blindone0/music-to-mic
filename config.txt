# Music to mic settings (changes are picked up within 2 s while running)
# music level in the call, dB RMS. Speech in Discord sits around -18..-20 dB, so -24 keeps music just under the voices.
# Raise to -20 for louder music, set 0 to disable auto-levelling (then music_gain is a fixed multiplier).
music_target_db=-24
# maximum boost the auto-leveller may apply (or the fixed gain when music_target_db=0)
music_gain=4
# your own voice level
mic_gain=1.0
# this program's sound is NOT sent to the mic, so people in the call never hear themselves
exclude_process=Discord.exe
# which microphone carries your voice (name or part of it). Empty = the mic that was default when you switched ON.
# A noise-cancelling virtual mic (NVIDIA Broadcast, Krisp, SteelSeries Sonar...) stops the speakers' sound leaking back = no echo.
mic=NVIDIA Broadcast
# VB-CABLE device names (leave as is). Several alternatives can be separated with |
cable_playback=CABLE Input|Speakers (VB-Audio Virtual Cable)
cable_recording=CABLE Output
# 1 = write mic/music levels to the log every 5 s (for troubleshooting)
verbose=0
